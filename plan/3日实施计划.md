### AgentFSM 核心

```python
class AgentFSM:
    states = ["init", "agent_think", "tool_call", "finish", "error"]

    def __init__(self, session_id: str, max_steps: int = 15):
        self.session_id = session_id
        self.current = "init"
        self.step = 0
        self.max_steps = max_steps

    def transit(self, to_state: str) -> bool:
        valid = {
            "init": {"agent_think"},
            "agent_think": {"tool_call", "finish", "error"},
            "tool_call": {"agent_think", "error"},
            "error": {"agent_think", "finish"},
        }
        if to_state not in valid.get(self.current, set()):
            return False
        self.current = to_state
        if to_state in ("tool_call", "agent_think"):
            self.step += 1
        return True

    def to_event(self) -> dict:
        return {
            "type": "fsm",
            "data": {
                "state": self.current,
                "step": self.step,
                "max_steps": self.max_steps,
            },
        }
```

### SSE 集成（stream_agent_events 中）

fsm = AgentFSM(session_id)

### ToolResult 定义

```python
@dataclass
class ToolResult:
    code: int
    data: Any = None
    error: str = ""
    extra: dict = field(default_factory=dict)

    @property
    def is_ok(self) -> bool:
        return self.code == 0

    def to_llm_text(self) -> str:
        """格式化成 LLM 可读的文本（兼容旧格式）"""
        if self.is_ok:
            return json.dumps({"ok": True, "data": self.data}, ensure_ascii=False)
        return json.dumps({"ok": False, "error": self.error}, ensure_ascii=False)
```

### 迁移策略

- `_ok`/`_err` 保留别名，内部改为调用 `ToolResult`
- 新工具直接返回 `ToolResult`
- 旧工具逐步迁移，每次改一个

### 网关骨架

```python
class ToolGateway:
    def __init__(self):
        self._entries: dict[str, ToolEntry] = {}
        self._middlewares: list[Middleware] = []

    def register(self, tool: BaseTool, sensitive: bool = False):
        self._entries[tool.name] = ToolEntry(tool=tool, sensitive=sensitive)

    async def call(self, name: str, params: dict, ctx: dict) -> ToolResult:
        for mw in self._middlewares:
            r = await mw.process(name, params, ctx)
            if r is not None:
                return r
        entry = self._entries.get(name)
        if not entry:
            return ToolResult(code=404, error=f"tool {name} not found")
        return await entry.tool.execute(params)
```

**本日暂不做的**：参数自动校验、限流中间件（留后续迭代）

**动态组装逻辑**：

```python
def build_prompt(user_input: str, tools: list) -> str:
    sections = [
        AgentPromptBuilder.MODULES["role"],
        AgentPromptBuilder.MODULES["tools_summary"],
        AgentPromptBuilder.MODULES["rules_common"],
    ]
    # 根据用户输入动态添加意图模块
    if any(kw in user_input for kw in ["生成简历", "做简历", "写简历"]):
        sections.append(AgentPromptBuilder.MODULES["intent_generate_resume"])
    if any(kw in user_input for kw in ["修改知识库", "改名", "换经历"]):
        sections.append(AgentPromptBuilder.MODULES["intent_knowledge_edit"])
    if any(kw in user_input for kw in ["创建面试", "新增面试", "录入面试"]):
        sections.append(AgentPromptBuilder.MODULES["intent_interview_record"])
    if "[文件:" in user_input:
        sections.append(AgentPromptBuilder.MODULES["file_upload"])
    return "\n\n".join(sections)
```

| 风险 | 概率 | 应对 |
